function buildPrompt(userPrompt, style) {

  let corrected = userPrompt;

  if (userPrompt.includes("kabba") || userPrompt.includes("khana kabba")) {
    corrected = "Kaaba in Mecca, Masjid al-Haram";
  }

  const styleMap = {
    realistic: "photorealistic, natural lighting",
    cinematic: "cinematic lighting, dramatic shadows",
    anime: "anime style, vibrant colors",
    "3d": "3D render, octane render"
  };

  const negative = "blurry, low quality, watermark, distorted, bad anatomy";

  return `${corrected}, ${styleMap[style]}, ultra detailed, sharp focus, ${negative}`;
}
