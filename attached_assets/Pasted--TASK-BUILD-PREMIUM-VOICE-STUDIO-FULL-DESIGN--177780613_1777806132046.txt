SECTION 7: AUDIO GENERATION ENGINE
--------------------------------------------------

Use FREE API:

https://api.streamelements.com/kappa/v2/speech

--------------------------------------------------

JS LOGIC:

function generateAudio() {

  const text = document.getElementById("script").value;
  const voice = getSelectedVoice();

  const voiceMap = {
    "Deep Narrator": "Brian",
    "Calm Therapist": "Amy",
    "News Anchor": "Matthew"
  };

  const apiVoice = voiceMap[voice] || "Brian";

  const url = `https://api.streamelements.com/kappa/v2/speech?voice=${apiVoice}&text=${encodeURIComponent(text)}`;

  const audio = document.getElementById("audioPlayer");
  audio.src = url;
  audio.play();
}
